name: Mise à jour des données

on:
  schedule:
    - cron: "17 5 * * *"      # tous les jours vers 6 h / 7 h (heure belge)
  workflow_dispatch:           # bouton « Run workflow » pour lancer à la main

permissions:
  contents: write

jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Télécharger les données
        run: python scripts/update_data.py
      - name: Enregistrer les données
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add data
          git diff --cached --quiet || git commit -m "Mise à jour des données"
          git push
